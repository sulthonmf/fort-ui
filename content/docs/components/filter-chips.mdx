export function Example() {
  const [filter, setFilter] = React.useState('all');
  const items = [
    { id: 'all', label: 'All Items' },
    { id: 'recent', label: 'Recently Added' },
    { id: 'favorites', label: 'Favorites' },
    { id: 'shared', label: 'Shared with Me' },
  ];

  return (
    <FilterChips
      items={items}
      value={filter}
      onValueChange={setFilter}
    />
  );
}`}
